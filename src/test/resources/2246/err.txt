public RoleListDto getRole(String adlogin, String channel, String sessionId, String traceId) {
        String uri = String.format(integrationUrlGetRole, adlogin, channel);
        uri = tsrmIntegrationServiceUrl+uri;
        ResponseEntity<RoleListDto> tsrmResponse;

        HttpEntity<Void> request =
                new HttpEntity<>(ClientHttpHeaders.minimal(adlogin, sessionId, traceId));
        tsrmResponse = loggingRestTemplate.exchange(uri, HttpMethod.GET, request, RoleListDto.class);

        // после вызова exchange в MDC затирается traceId
        MDC.put("traceId", traceId);

        if (tsrmResponse.hasBody()) {
                return tsrmResponse.getBody();
        } else {
            return null;
        }
    }
