@Data
@AllArgsConstructor
@Schema(description = "Массив параметров деталей ошибки от сервиса", name = "ServiceError")
public class ServiceError {
  @Schema(description = "Текст ошибки", example = "Ответ не найден", maxLength = 4000, required = false, nullable = true)
  private String message;
  @Schema(description = "Код ошибки", example = "", maxLength = 200, required = false, nullable = true)
  private String code;
  @Schema(description = "Трейс ошибки", example = "", maxLength = 200, required = false, nullable = true)
  private String logId;
  @ArraySchema(
      maxItems = 99999,
      uniqueItems = true,
      arraySchema = @Schema(
          description = "Массив с детализированными данными по ошибке от сервисов",
          example = "",
          required = false, nullable = true
      )
  )
  private List<SapErrorDetails> details;

  public ServiceError() {
    this.message = "Дефолтный текст ошибки";
    this.code = "";
    this.logId = "";
    this.details = new ArrayList<>();
  }

  public ServiceError(String message) {
    this.message = message;
    this.code = "";
    this.logId = "";
    this.details = new ArrayList<>();
  }
}    


@Data
@AllArgsConstructor
@NoArgsConstructor
@Schema(description = "Массив параметров деталей ошибки от сап", name = "SapErrorDetails")

public class SapErrorDetails {
    @Schema(description = "Код ошибки", example = "'510'", maxLength = 255, required = false, nullable =  true)
    private String code;
    @Schema(description = "Текст ошибки", example = "Отсутствуют полномочия к данной группе", maxLength = 1000, required = false, nullable =  true)
    private String message;
    @Schema(description = "Уровень критичности", example = "Error", maxLength = 255, required = false, nullable =  true)
    @JsonProperty("@SAP__common.Severity")
    private String severity;
}
