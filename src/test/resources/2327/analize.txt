@AllArgsConstructor
@Data
@Schema(name = "LkrError", description = "Тело ошибки")
public class LkrError {

    @Schema(description = "Текст ошибки", example = "Ответ не найден", maxLength = 4000, required = false, nullable = true)
    private String message;

    @Schema(description = "Идентификатор трассировки ошибки", example = "", maxLength = 200, required = false, nullable = true)
    private String logId;

    @ArraySchema(
            maxItems = 99999,
            uniqueItems = true,
            arraySchema = @Schema(
                    description = "Массив с детализированными данными по ошибке",
                    example = "",
                    required = false, nullable = true
            )
    )
    private List<LkrErrorDetail> details;

    public LkrError() {
        message = "";
        details = new ArrayList<>();
    }


}

@AllArgsConstructor
@Data
@Schema(name = "LkrErrorDetail", description = "Детальная информация по ошибке")
public class LkrErrorDetail {
    @Schema(description = "Код ошибки", example = "'510'", maxLength = 255, required = false, nullable = true)
    private String code;

    @Schema(description = "Текст ошибки", example = "Отсутствуют полномочия к данной группе", maxLength = 1000, required = false, nullable = true)
    private String message;

    public LkrErrorDetail() {
        code = "";
        message = ErrorResponse.DEFAULT_ERROR_MESSAGE;
    }
}
