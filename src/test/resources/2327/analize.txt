catch (HttpClientErrorException | HttpServerErrorException e) {

protected void handleHttpClientOrServerErrorException(HttpStatusCodeException e, SapResponseDtoExtended dto, SapSystemParamsDto systemParams, String errorMessage) {
        dto.setResponseCode(e.getStatusCode().toString());
        dto.setErrorMessage(e.getMessage());
        entityService.updateRequestStatus(systemParams.getUuid(), RequestStatusConstant.ERROR_REPLY);
        log.warn(errorMessage + ": [{}]", e.getMessage());
        log.debug("Стектрейс: \n" + ExceptionUtils.getStackTrace(e));
    }


private String finishProcessingMessage(Long start, SapSystemParamsDto systemParams, SapResponseDtoExtended dto, RequisitionRequestEntity requestEntity, String code) throws Exception {
        long end = System.currentTimeMillis();
        long processingTime = end - start;

        checkDtoUuid(systemParams, dto);

        dto.setRolesList(systemParams.getRolesHeader());
        dto.setTabNumber(systemParams.getTabNumber());


        String messageOutcoming = mapper.writeValueAsString(dto);
        RequisitionResponseEntity responseEntity = saveAndGetResponse(systemParams.getUuid(), requestEntity, messageOutcoming, processingTime, code);

        if (code.equals("200")) {
            entityService.deleteResponseByUidAndId(systemParams.getUuid(), responseEntity.getId().toString());
            entityService.deleteRequestByUidAndId(systemParams.getUuid(), requestEntity.getId().toString());
        }

        messageOutcoming = checkMessageSize(dto, messageOutcoming);
        return messageOutcoming;
    }
