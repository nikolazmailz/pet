    private ServiceErrorDto getSapErrorDescription(String sapError) {
        try {
            int start = sapError.indexOf("{\"error\"");
            String errorText = sapError.substring(start);
            ObjectMapper mapper = new ObjectMapper();
            SapErrorResponse sapErrorResponse = mapper.readValue(errorText, SapErrorResponse.class);
            ServiceError error = new ServiceError();
            error.setMessage(sapErrorResponse.getError().getMessage());
            error.setDetails(sapErrorResponse.getError().getDetails());
            return new ServiceErrorDto(error);
        } catch (Exception e) {
            return new ServiceErrorDto();
        }
    }

    @ExceptionHandler(value = Exception.class)
    public ResponseEntity<Object>
    defaultErrorHandler(HttpServletRequest req, Exception e) throws Exception {
        String requestType;
        String sessionId;
        String pernr;
        String adlogin;
        String traceId = (req.getAttribute("traceId") != null) ? req.getAttribute("traceId").toString() : "undefined";
        ServiceErrorDto serviceError = new ServiceErrorDto();
        LocalDateTime sessionCreatedAt = LocalDateTime.now();
        if (e.getMessage().contains("SAP")) {
            serviceError = getSapErrorDescription(e.getMessage());
        } else {
            serviceError = serviceError.setError(new ServiceError(e.getMessage(), "500", traceId, new ArrayList<>()));
        }
        if (req.getAttribute("requestType") != null && req.getAttribute("sessionId") != null && req.getAttribute("pernr") != null && req.getAttribute("adlogin") != null && req.getAttribute("traceId") != null) {
            requestType = req.getAttribute("requestType").toString();
            sessionId = req.getAttribute("sessionId").toString();
            pernr = req.getAttribute("pernr").toString();
            adlogin = req.getAttribute("adlogin").toString();
            traceId = req.getAttribute("traceId").toString();
        } else {
            log.warn(ErrorResponse.ERROR_NOT_ALL_REQUIRED_AUDIT_PARAMS_MESSAGE);
            requestType = (req.getAttribute("requestType") != null) ? req.getAttribute("requestType").toString() : "undefined";
            sessionId = (req.getAttribute("sessionId") != null) ? req.getAttribute("sessionId").toString() : "undefined";
            pernr = (req.getAttribute("pernr") != null) ? req.getAttribute("pernr").toString() : "undefined";
            adlogin = (req.getAttribute("adlogin") != null) ? req.getAttribute("adlogin").toString() : "undefined";
            traceId = (req.getAttribute("traceId") != null) ? req.getAttribute("traceId").toString() : "undefined";
        }
        LogUtil.logSapErrorResponse(pernr, sessionId, e.getMessage(), traceId, null);
        sendUnexpectedErrorAuditEvent(requestType, sessionId, sessionCreatedAt, serviceError.getError().getMessage(), pernr, adlogin);
        return new ResponseEntity<>(serviceError, HttpStatus.INTERNAL_SERVER_ERROR);
    }

    @ExceptionHandler(value = HandlerMethodValidationException.class)
    public ResponseEntity<Object>
    validationMethodErrorHandler(HandlerMethodValidationException e) {
        String validationErrors = "";
        for (ParameterValidationResult validationResult : e.getAllValidationResults()) {
            List<MessageSourceResolvable> validationErrorList = validationResult.getResolvableErrors();
            validationErrors = validationErrors.concat(validationErrorList.stream().map(MessageSourceResolvable::toString)
                    .collect(Collectors.joining(", ")));
        }
        ServiceError error = new ServiceError(validationErrors, String.valueOf(e.getStatusCode()), "undefined", new ArrayList<>());
        ServiceErrorDto errorDto = new ServiceErrorDto(error);
        return new ResponseEntity<>(errorDto, HttpStatus.UNPROCESSABLE_ENTITY);
    }
