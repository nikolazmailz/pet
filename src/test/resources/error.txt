import jakarta.persistence.AttributeConverter;
import jakarta.persistence.Converter;
import java.time.Duration;

@Converter
public class DurationToIntervalConverter implements AttributeConverter<Duration, String> {

    @Override
    public String convertToDatabaseColumn(Duration duration) {
        return duration == null ? null : duration.getSeconds() + " seconds";
    }

    @Override
public Duration convertToEntityAttribute(String dbData) {
    if (dbData == null) return null;
    String[] parts = dbData.split(":");
    if (parts.length != 3) {
        throw new IllegalArgumentException("Unexpected INTERVAL format: " + dbData);
    }
    return Duration.ofHours(Long.parseLong(parts[0]))
            .plusMinutes(Long.parseLong(parts[1]))
            .plusSeconds(Long.parseLong(parts[2]));
}
}
