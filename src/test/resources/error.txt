    @Test
    void shouldReturnCorrectCount_WhenNoFilters() {
        // Given
        Integer vacancyId = vacancyRepository.findAll().stream()
                .findFirst()
                .map(VacancyEntity::getId)
                .orElse(null);

        String search = null;
        String[] eventCodeList = {};
        String[] pairEventCodes = {};
        String[] pairStatusCodes = {};

        // When
        Integer result = repository.countActiveCandidatesAndCandidatesEventsByVacancyIdWithFilters(
                vacancyId, search, eventCodeList, pairEventCodes, pairStatusCodes);

        // Then: должны быть 2 кандидата с последними активными событиями
        assertThat(result).isEqualTo(2);
    }

@Query(value = "with active_check as ( select ce.candidate_id, cst.is_active FROM srhr_requisition.candidates_events ce  " +
            "JOIN srhr_requisition.candidates c ON ce.candidate_id = c.id   " +
            "JOIN srhr_requisition.candidate_stage cs on ce.event_code = cs.code  " +
            "JOIN srhr_requisition.candidate_stage_status css on cs.id = css.stage_id  " +
            "JOIN srhr_requisition.candidate_status cst on css.status_id = cst.id  " +
            "WHERE c.vacancy_id = :vacancyId AND ce.status_code = cst.code  " +
            ") SELECT count(ce.*)   " +
            "FROM srhr_requisition.candidates_events ce  " +
            "JOIN srhr_requisition.candidates c ON ce.candidate_id = c.id  " +
            "WHERE c.vacancy_id = :vacancyId  " +
            "  AND ce.creation_date = (  " +
            "      SELECT MAX(ce2.creation_date)  " +
            "      FROM srhr_requisition.candidates_events ce2  " +
            "      JOIN srhr_requisition.candidates c2 ON ce2.candidate_id = c2.id  " +
            "      WHERE c2.vacancy_id = :vacancyId  " +
            "        AND ce2.candidate_id = ce.candidate_id   " +
            "  ) AND ce.candidate_id in (select candidate_id from active_check where is_active = 1)  " +
            "  AND ce.status_code in (select cs.code from srhr_requisition.candidate_status cs where cs.is_active = 1)   " +
            "  AND (" +
            "    cardinality(:pairStatusCodes) = 0 " +                       // пар нет -> фильтр по чистым event
            "      AND (cardinality(:eventCodeList) = 0 OR ce.event_code = ANY(:eventCodeList)) " +
            "    OR " +
            "    cardinality(:pairStatusCodes) > 0 " +                       // пары есть -> матчим (event, status)
            "      AND (ce.event_code, ce.status_code) IN (" +
            "        SELECT e, s FROM unnest(:pairEventCodes, :pairStatusCodes) AS pair(e, s)" +
            "      )" +
            "  ) " +
            "            AND (cast(:search as text) IS NULL or c.full_name ILIKE :search)", nativeQuery = true)
    Integer countActiveCandidatesAndCandidatesEventsByVacancyIdWithFilters (
            @Param("vacancyId") Integer vacancyId,
            @Param("search") String search,
            @Param("eventCodeList") String[] eventCodeList,
            @Param("pairEventCodes") String[] pairEventCodes,
            @Param("pairStatusCodes") String[] pairStatusCodes
