@Configuration
@Slf4j
public class TaskConfiguration {

    @Value("${app.scheduler.core-pool-size}")
    private int corePoolSize;

    @Value("${app.scheduler.max-pool-size}")
    private int maxPoolSize;

    @Value("${app.scheduler.queue-capacity}")
    private int queueCapacity;

    @Bean(name = "ProducerExecutor")public Executor ThreadPoolTaskExecutor(){
        ThreadPoolTaskExecutor threadPoolTaskExecutor = new ThreadPoolTaskExecutor();
        threadPoolTaskExecutor.setCorePoolSize(corePoolSize);
        threadPoolTaskExecutor.setMaxPoolSize(maxPoolSize);
        threadPoolTaskExecutor.setQueueCapacity(queueCapacity);
        threadPoolTaskExecutor.setThreadNamePrefix("producer-ex-");
        threadPoolTaskExecutor.setWaitForTasksToCompleteOnShutdown(true);
        log.info("Конфигурируем ThreadPoolTaskExecutor");
        return threadPoolTaskExecutor;
    }
}



    private final RedisAdapter redisAdapter;
    private final TaskCleanerRepositoryImpl taskCleanerRepository;
    private final TaskScheduleRepository taskScheduleRepository;
    @Value("${spring.data.redis.queue.name.housekeep}") String housekeepRequestQueue;
    @Scheduled(cron = "${app.scheduler.db.cron.cleaner}")
    @Async(value = "ProducerExecutor")
    public void produce() throws JsonProcessingException {
        int count = 100;
        for (int i = 0; i < count; i++) {
            String request = redisAdapter.getFromQueue(housekeepRequestQueue);
            if (request!=null) {
                try {
                    ObjectMapper mapper = new ObjectMapper();
                    mapper.registerModule(new JavaTimeModule());
                    mapper.disable(SerializationFeature.WRITE_DATES_AS_TIMESTAMPS);
                    ScheduledTaskDto dto = mapper.readValue(request, ScheduledTaskDto.class);
                    taskCleanerRepository.deleteData(dto.getSchemaName(), dto.getTableName(), dto.getConditionColumn(), dto.getDataExpirationInHours());
                    taskScheduleRepository.updatelastSuccessfulCleanDateById(dto.getTaskId(), LocalDateTime.now());
                } catch (Exception e) {
                    log.error("Произошла ошибка в таске очистки кэша бд [CleanerTask]. Текст ошибки: [{}]", e.getMessage());
                }
            }
        }
    }
