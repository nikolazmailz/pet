private static CandidateStageEntity stage(String code, Integer sorting) {
        CandidateStageEntity s = new CandidateStageEntity();
        s.setCode(code);
        s.setSorting(sorting);
        return s;
    }

    private static CandidateStatusNameProjection projection(String eventCode,
                                                            String statusCode,
                                                            String name) {
        return new CandidateStatusNameProjection() {
            @Override public String getEventCode() { return eventCode; }
            @Override public String getStatusCode() { return statusCode; }
            @Override public String getName() { return name; }
        };
    }
