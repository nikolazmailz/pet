/**
 * Один пул соединений и один {@link org.springframework.web.client.RestTemplate} на роль,
 * живут всё время работы приложения.
 * Заголовки, зависящие от запроса (pernr, sessionId, csrf, cookie...),
 * собираются через requestHeaders и передаются в RequestEntity
 */
@Slf4j
@Component
public class EasupClient {

    public static final String HEADER_PERNR = "pernr";
    public static final String HEADER_SESSION_ID = "sessionId";
    public static final String HEADER_CSRF_TOKEN = "x-csrf-token";
    public static final String HEADER_ROLES = "Roles";
    public static final String HEADER_LOG_ID = "logId";

    public static final String CSRF_FETCH = "fetch";

    private static final String CONTENT_TYPE_JSON_UTF8 = "application/json;charset=UTF-8";

    private static final int MAX_CONNECTIONS_TOTAL = 50;
    private static final int MAX_CONNECTIONS_PER_ROUTE = 20;
    private static final TimeValue CONNECTION_TTL = TimeValue.ofMinutes(5);
    private static final TimeValue VALIDATE_AFTER_INACTIVITY = TimeValue.ofSeconds(10);
    private static final TimeValue IDLE_EVICTION = TimeValue.ofMinutes(1);

    private final EasupCertificateConfig certificateConfig;
    private final EasupClientSideMetricsInterceptor metricsInterceptor;
    private final EasupProperties.Headers headerProps;

    private final String basicAuthHeaderValue;

    private final Map<String, CloseableHttpClient> httpClientsByRole = new ConcurrentHashMap<>();
    private final Map<String, RestTemplate> restTemplatesByRole = new ConcurrentHashMap<>();

    public EasupClient(EasupCertificateConfig certificateConfig,
                       EasupClientSideMetricsInterceptor metricsInterceptor,
                       EasupProperties properties) {
        this.certificateConfig = certificateConfig;
        this.metricsInterceptor = metricsInterceptor;
        this.headerProps = properties.getHeaders();
        this.basicAuthHeaderValue = buildBasicAuth(properties.getBasicAuth());
    }


    /** Общий RestTemplate для роли. Потокобезопасен, создаётся один раз на роль. */
    public RestTemplate forRole(String role) {
        return restTemplatesByRole.computeIfAbsent(role, r -> new RestTemplateBuilder()
                .requestFactory(() -> new HttpComponentsClientHttpRequestFactory(httpClient(r)))
                .additionalInterceptors(new EasupRequestInterceptor(), metricsInterceptor)
                .build());
    }

    /** Заголовки запроса без CSRF-токена */
    public HttpHeaders requestHeaders(SapSystemParamsDto params) {
        return buildHeaders(params, CSRF_FETCH, List.of());
    }

    /**
     * Заголовки запроса с CSRF-токеном: клиент делает GET,
     * забирает из ответа x-csrf-token и Set-Cookie и подставляет их в заголовки.
     */
    public HttpHeaders csrfHeaders(SapSystemParamsDto params, URI csrfUrl) {
        RequestEntity<Void> fetch = RequestEntity.get(csrfUrl)
                .headers(requestHeaders(params))
                .build();
        HttpHeaders response = forRole(params.getRole()).exchange(fetch, Void.class).getHeaders();

        String token = response.getFirst(HEADER_CSRF_TOKEN);
        if (!StringUtils.hasText(token)) {
            throw new HttpClientErrorException(HttpStatus.UNAUTHORIZED,
                    "Не были получены токен и куки для CSRF аутентификации.");
        }
        List<String> cookies = response.getOrDefault(HttpHeaders.SET_COOKIE, List.of()).stream()
                .map(EasupClient::cookieNameValue)
                .toList();
        return buildHeaders(params, token, cookies);
    }

    /**
     * Заголовки запроса с уже известными токеном и куками.
     */
    public HttpHeaders requestHeaders(SapSystemParamsDto params, Map<String, List<String>> csrfHeaders) {
        List<String> csrf = csrfHeaders.get(HEADER_CSRF_TOKEN);
        if (csrf == null || csrf.isEmpty()) {
            throw new IllegalStateException("Не передан заголовок " + HEADER_CSRF_TOKEN);
        }
        return buildHeaders(params, csrf.get(0), csrfHeaders.getOrDefault("cookie", List.of()));
    }

    // legacy API

    @Deprecated
    public RestTemplate buildRestTemplateWithRoles(SapSystemParamsDto params) {
        return withDefaultHeaders(params.getRole(), requestHeaders(params));
    }

    @Deprecated
    public RestTemplate buildRestTemplateWithCSRFTokenWithRoles(SapSystemParamsDto params,
                                                                Map<String, List<String>> csrfHeaders) {
        return withDefaultHeaders(params.getRole(), requestHeaders(params, csrfHeaders));
    }

    private RestTemplate withDefaultHeaders(String role, HttpHeaders headers) {
        RestTemplateBuilder builder = new RestTemplateBuilder()
                .requestFactory(() -> new HttpComponentsClientHttpRequestFactory(httpClient(role)))
                .additionalInterceptors(new EasupRequestInterceptor(), metricsInterceptor);
        for (Map.Entry<String, List<String>> header : headers.entrySet()) {
            builder = builder.defaultHeader(header.getKey(), header.getValue().toArray(String[]::new));
        }
        return builder.build();
    }

    // internals

    private HttpHeaders buildHeaders(SapSystemParamsDto params, String csrfToken, List<String> cookies) {
        HttpHeaders headers = new HttpHeaders();
        headers.add(HttpHeaders.CONTENT_TYPE, CONTENT_TYPE_JSON_UTF8);
        if (basicAuthHeaderValue != null) {
            headers.add(HttpHeaders.AUTHORIZATION, basicAuthHeaderValue);
        }
        headers.add(HEADER_PERNR, params.getTabNumber());
        headers.add(HEADER_SESSION_ID, params.getSessionId());
        headers.add(HEADER_CSRF_TOKEN, csrfToken);
        headers.add(HEADER_ROLES, params.getRolesHeader());
        headers.add(HEADER_LOG_ID, params.getTraceId());
        headers.add(headerProps.getServiceIdHeaderName(), serviceIdFor(params.getRealm()));
        cookies.forEach(cookie -> headers.add(HttpHeaders.COOKIE, cookie));
        return headers;
    }

    private static String cookieNameValue(String setCookie) {
        return setCookie.split(";", 2)[0].trim();
    }

    private CloseableHttpClient httpClient(String role) {
        return httpClientsByRole.computeIfAbsent(role, this::createHttpClient);
    }

    private CloseableHttpClient createHttpClient(String role) {
        SSLContext sslContext = certificateConfig.configureHttpClientSsl(role);
        SSLConnectionSocketFactory sslSocketFactory =
                new SSLConnectionSocketFactory(sslContext, NoopHostnameVerifier.INSTANCE);

        PoolingHttpClientConnectionManager connectionManager = PoolingHttpClientConnectionManagerBuilder.create()
                .setSSLSocketFactory(sslSocketFactory)
                .setMaxConnTotal(MAX_CONNECTIONS_TOTAL)
                .setMaxConnPerRoute(MAX_CONNECTIONS_PER_ROUTE)
                .setDefaultConnectionConfig(ConnectionConfig.custom()
                        .setTimeToLive(CONNECTION_TTL)
                        .setValidateAfterInactivity(VALIDATE_AFTER_INACTIVITY)
                        .build())
                .build();

        log.info("Создан пул соединений ЕАСУП для роли [{}]", role);
        return HttpClients.custom()
                .setConnectionManager(connectionManager)
                .disableCookieManagement()
                .evictExpiredConnections()
                .evictIdleConnections(IDLE_EVICTION)
                .build();
    }

    private String serviceIdFor(String realm) {
        return RealmConstant.STAFF_MOBILE_REALM.equals(realm)
                ? headerProps.getServiceIdHeaderStaffMobileValue()
                : headerProps.getServiceIdHeaderStaffValue();
    }

    private static String buildBasicAuth(EasupProperties.BasicAuth auth) {
        if (!auth.isEnable() || !StringUtils.hasText(auth.getLogin()) || !StringUtils.hasText(auth.getPassword())) {
            return null;
        }
        String credentials = auth.getLogin() + ":" + auth.getPassword();
        return "Basic " + Base64.getEncoder().encodeToString(credentials.getBytes(StandardCharsets.ISO_8859_1));
    }

    @PreDestroy
    void shutdown() {
        httpClientsByRole.forEach((role, client) -> {
            try {
                client.close();
            } catch (IOException e) {
                log.warn("Не удалось закрыть HttpClient ЕАСУП для роли [{}]", role, e);
            }
        });
    }
}



package ru.ntdev.srhr.ms.master.data.integration.config.epa;

import java.io.FileInputStream;
import java.io.IOException;
import java.io.InputStream;
import java.security.GeneralSecurityException;
import java.security.KeyStore;
import java.util.Map;
import java.util.concurrent.ConcurrentHashMap;
import javax.net.ssl.SSLContext;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.apache.hc.client5.http.ssl.TrustAllStrategy;
import org.apache.hc.core5.ssl.SSLContextBuilder;
import org.springframework.stereotype.Component;
import org.springframework.util.ResourceUtils;
import ru.ntdev.srhr.common.constant.RolesConstant;

@Slf4j
@Component
@RequiredArgsConstructor
public class EasupCertificateConfig {

    /** Роль из запроса → ключ в easup.ssl.* */
    private static final Map<String, String> ROLE_TO_STORE_KEY = Map.of(
            RolesConstant.MSS_FULL, "mss-full",
            RolesConstant.MSS_LIMITED, "mss-limited",
            RolesConstant.MSS_STANDARD, "mss-standard",
            RolesConstant.ESS, "ess",
            RolesConstant.TABEL, "tabel"
    );

    private final EasupProperties properties;
    private final Map<String, SSLContext> contextsByRole = new ConcurrentHashMap<>();

    public SSLContext configureHttpClientSsl(String role) {
        String storeKey = ROLE_TO_STORE_KEY.get(role);
        if (storeKey == null) {
            log.warn("При выборе сертификата не найдена указанная в запросе роль: [{}]", role);
            throw new IllegalArgumentException(
                    "Указанная роль не числится среди вариантов при выборе сертификата: " + role);
        }
        return contextsByRole.computeIfAbsent(role, r -> buildContext(storeKey));
    }

    private SSLContext buildContext(String storeKey) {
        EasupProperties.SslStores stores = properties.getSsl().get(storeKey);
        if (stores == null) {
            throw new IllegalStateException("Не задана конфигурация easup.ssl." + storeKey);
        }
        try {
            KeyStore trustStore = loadKeyStore(stores.getTrustStore(), stores.getTrustStorePassword(), stores.getTrustStoreType());
            KeyStore keyStore = loadKeyStore(stores.getKeyStore(), stores.getKeyStorePassword(), stores.getKeyStoreType());

            return SSLContextBuilder.create()
                    .loadKeyMaterial(keyStore, stores.getKeyStorePassword().toCharArray())
                    .loadTrustMaterial(trustStore, TrustAllStrategy.INSTANCE)
                    .build();
        } catch (GeneralSecurityException | IOException e) {
            throw new IllegalStateException("Ошибка сборки SSLContext для easup.ssl." + storeKey, e);
        }
    }

    private static KeyStore loadKeyStore(String path, String password, String type)
            throws GeneralSecurityException, IOException {
        KeyStore keyStore = KeyStore.getInstance(type);
        try (InputStream in = new FileInputStream(ResourceUtils.getFile(path))) {
            keyStore.load(in, password.toCharArray());
        }
        return keyStore;
    }
}

package ru.ntdev.srhr.ms.master.data.integration.config.epa.filter;

import java.io.IOException;
import org.springframework.http.HttpHeaders;
import org.springframework.http.HttpRequest;
import org.springframework.http.client.ClientHttpRequestExecution;
import org.springframework.http.client.ClientHttpRequestInterceptor;
import org.springframework.http.client.ClientHttpResponse;
import ru.ntdev.srhr.common.util.LogUtil;

public class EasupRequestInterceptor implements ClientHttpRequestInterceptor {


    public EasupRequestInterceptor() {

    }

    @Override
    public ClientHttpResponse intercept(
            HttpRequest request,
            byte[] body,
            ClientHttpRequestExecution execution
    ) throws IOException {

        HttpHeaders headers = request.getHeaders();
        LogUtil.logRequest(headers.get("pernr").toString(), headers.get("sessionId").toString(), request.getMethod().name(), request.getURI().toString(), true,
                headers, headers.get("logId").toString());
        ClientHttpResponse response = execution.execute(request, body);
        if (response.getStatusCode().is2xxSuccessful()) {
            LogUtil.logResponse(headers.get("pernr").toString(), headers.get("sessionId").toString(), request.getMethod().name(), request.getURI().toString(), String.valueOf(response.getStatusCode().value()), null, true, response.getHeaders(), headers.get("logId").toString());
        } else {
            LogUtil.logResponse(headers.get("pernr").toString(), headers.get("sessionId").toString(), request.getMethod().name(), request.getURI().toString(), String.valueOf(response.getStatusCode().value()), null, true, response.getHeaders(), headers.get("logId").toString());
        }
        return response;
    }
}


package ru.ntdev.srhr.ms.master.data.integration.config.epa.filter;

import io.micrometer.core.instrument.MeterRegistry;
import io.micrometer.core.instrument.Tags;
import io.micrometer.core.instrument.Timer;
import java.io.IOException;
import java.util.concurrent.TimeUnit;
import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Component;

@Component
@Slf4j
public class EasupRequestDurationTimer {
    @Autowired
    private MeterRegistry registry;

    private static final String TIMER_NAME = "http_client_requests_seconds_count";
    private static final String TIMER_LABEL_SYSTEM = "system";
    private static final String TIMER_LABEL_ENDPOINT = "endpoint";
    private static final String TIMER_LABEL_METHOD = "method";
    private static final String TIMER_LABEL_STATUS = "status";
    private static final String SYSTEM_NAME = "easup";

    private Timer timer;

    public void setTimer(String endpoint, String method, String responseCode) throws IOException {
        timer = Timer.builder(TIMER_NAME)
                .tags(createTags(
                        SYSTEM_NAME,
                        endpoint,
                        method,
                        responseCode))
                .register(registry);
    }

    public void record(long duration) {
        timer.record(duration, TimeUnit.MILLISECONDS);
    }

    private Tags createTags(String system, String endpoint, String method, String status) {
        return Tags.of(
                TIMER_LABEL_SYSTEM, system,
                TIMER_LABEL_ENDPOINT, endpoint,
                TIMER_LABEL_METHOD, method,
                TIMER_LABEL_STATUS, status
        );
    }

}


package ru.ntdev.srhr.ms.master.data.integration.config.epa.filter;

import java.io.IOException;
import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.http.HttpRequest;
import org.springframework.http.client.ClientHttpRequestExecution;
import org.springframework.http.client.ClientHttpRequestInterceptor;
import org.springframework.http.client.ClientHttpResponse;
import org.springframework.stereotype.Component;

@Component
@Slf4j
public class EasupClientSideMetricsInterceptor implements ClientHttpRequestInterceptor {
    @Autowired
    private EasupRequestDurationTimer timer;

    private static final String ERROR_CODE = "500";

    public ClientHttpResponse intercept(HttpRequest request, byte[] body, ClientHttpRequestExecution execution) throws IOException {
        log.debug("Сработал перехватчик для подсчёта метрики timer");

        long start = 0;
        long end;
        ClientHttpResponse response;
        String endpoint = formatRequestEndpoint(request);
        String method = request.getMethod().name();

        try {
            start = System.currentTimeMillis();
            response = execution.execute(request, body);
        } catch (Exception e) {
            end = System.currentTimeMillis();
            timer.setTimer(endpoint, method, ERROR_CODE);
            timer.record(end - start);
            throw e;
        }

        end = System.currentTimeMillis();
        String statusCode = response.getStatusCode().toString().substring(0, 3);

        timer.setTimer(endpoint, method, statusCode);
        timer.record(end - start);

        return response;
    }

    private String formatRequestEndpoint(HttpRequest request) {
        String endpoint = request.getURI().getPath();
        int endEndpointIndex = endpoint.indexOf('(');

        if (endEndpointIndex != -1) {
            endpoint = endpoint.substring(0, endEndpointIndex);
        }

        return endpoint;
    }

}



 private void example() {
        // Чтение, без токена:
        RequestEntity<Void> request1 = RequestEntity.get("uri")
                .headers(easupClient.requestHeaders(params))
                .build();
        Object dto1 = easupClient.forRole(params.getRole())
                .exchange(request1, Object.class)
                .getBody();

        // Запись, клиент сам получает токен:
        RequestEntity<?> request2 = RequestEntity.post("uri")
                .headers(easupClient.csrfHeaders(params, URI.create("csrfUri")))
                .body(body);
        Object dto2 = easupClient.forRole(params.getRole())
                .exchange(request2, Object.class)
                .getBody();

        // Запись, токен и куки уже есть (мапа с ключами x-csrf-token и cookie):
        RequestEntity<?> request3 = RequestEntity.post("uri")
                .headers(easupClient.requestHeaders(params, csrfHeaders))
                .body(body);
        Object dto3 = easupClient.forRole(params.getRole())
                .exchange(request3, Object.class)
                .getBody();
    }
