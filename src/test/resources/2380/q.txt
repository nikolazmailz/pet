@Service
@RequiredArgsConstructor
public class EasupService {

    @Value("${sap.service-delegation-url}")
    private String serviceDelegationUrl;

    @Value("${tyk.host}")
    private String tykHost;

    private final EasupClient easupClient;


    public void create(SapSystemParamsDto systemParamsDto, String serviceId) {
        String spoUrl = String.format(serviceDelegationUrl);
        String uri = tykHost + spoUrl;

        RequestEntity<Void> request = RequestEntity.get(uri)
                .headers(easupClient.requestHeaders(systemParamsDto, serviceId))
                .build();

        easupClient.restTemplate().exchange(request, Void.class).getBody();

        RequestEntity<Void> requestPost = RequestEntity.post(uri)
                .headers(easupClient.requestHeaders(systemParamsDto, serviceId))
                .build();

        easupClient.restTemplate().exchange(requestPost, Void.class).getBody();
    }
}
